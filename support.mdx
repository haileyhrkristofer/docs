export const HubspotForm = ({ portalId, formId, region = "eu1" }) => {
  const targetId = `hs-form-${formId}`;

  useEffect(() => {
    const create = () => {
      const target = document.getElementById(targetId);
      if (!window.hbspt || !target) return;
      target.innerHTML = "";
      window.hbspt.forms.create({ portalId, formId, region, target: `#${targetId}`, css: "" });
    };

    if (window.hbspt) {
      create();
      return;
    }

    const src = `https://js-${region}.hsforms.net/forms/embed/v2.js`;
    let script = document.querySelector(`script[src="${src}"]`);
    if (!script) {
      script = document.createElement("script");
      script.src = src;
      script.charset = "utf-8";
      script.async = true;
      document.body.appendChild(script);
    }
    script.addEventListener("load", create);
    return () => script.removeEventListener("load", create);
  }, [portalId, formId, region]);

  return (
    <div className="hailey-hs-form not-prose">
      <style>{hsFormStyles}</style>
      <div id={targetId} style={{ minHeight: "300px" }} />
    </div>
  );
};
